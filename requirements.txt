# --- Phase 3: procedural retrieval ---
# CPU-only embeddings. No document text leaves the machine to build the
# index -- the same argument the Phase 4 hybrid LLM split makes, one layer
# down. This pulls torch (~200MB); it is the heaviest install in the repo.
# The lexical arm and the vector store are Postgres (pgvector), so there is
# no separate search service to run.
sentence-transformers==6.1.0
# Intel-macOS (x86_64) ceiling. PyTorch's last x86_64 macOS wheel is 2.2.2,
# which pins the rest: transformers 5.0.0 is the only 5.x (as
# sentence-transformers 6 requires) still accepting torch 2.2, and torch 2.2
# was built against numpy 1.x. The markers keep Linux (the Docker image)
# and Apple Silicon on current versions.
torch==2.2.2; sys_platform == "darwin" and platform_machine == "x86_64"
transformers==5.0.0; sys_platform == "darwin" and platform_machine == "x86_64"
numpy<2; sys_platform == "darwin" and platform_machine == "x86_64"
